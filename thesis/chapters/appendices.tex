\section{IMU Register Configuration}

\begin{lstlisting}[language=c]
// ICM-42688-P Register Initialization
// Reference: Datasheet Section 4.3

void imu_init(void) {
    // 1. Soft Reset
    imu_write_reg(REG_PWR_MODE, 0x80);
    delay(50);  // Wait for reset to complete

    // 2. Set Clock Source (internal oscillator)
    imu_write_reg(REG_CLK_CONFIG, 0x01);

    // 3. Configure Gyro
    // Range: +/-2000 dps, ODR: 1000 Hz, DLPF: enabled
    imu_write_reg(REG_GYRO_CONFIG0, 0x04);  // 4 = 2000 dps
    imu_write_reg(REG_GYRO_CONFIG1, 0x12);  // 1 kHz ODR, 176 Hz DLPF

    // 4. Configure Accelerometer
    // Range: +/-16g, ODR: 1000 Hz, DLPF: enabled
    imu_write_reg(REG_ACCEL_CONFIG0, 0x04);  // 4 = 16g
    imu_write_reg(REG_ACCEL_CONFIG1, 0x12);  // 1 kHz ODR, 484 Hz DLPF

    // 5. Configure FIFO
    imu_write_reg(REG_FIFO_CONFIG1, 0x08);   // FIFO mode enabled
    imu_write_reg(REG_FIFO_CONFIG2, 0x00);   // Sample count = 0 (unlimited)

    // 6. Configure Interrupts
    imm_write_reg(REG_INT_CONFIG0, 0x02);    // INT1 active high
    imu_write_reg(REG_INT_CONFIG1, 0x00);    // Push-pull output

    // 7. Enable FSYNC timestamping
    imu_write_reg(REG_FSYNC_CONFIG, 0x20);   // FSYNC in FIFO

    // 8. Set Operating Mode
    imu_write_reg(REG_PWR_MODE, 0x01);       // Run mode
}
\end{lstlisting}

\section{IMU Raw Data CSV Format}

\begin{verbatim}
unified_time_s,esp_timestamp_us,host_timestamp_us,
  accel_x_g,accel_y_g,accel_z_g,gyro_x_dps,gyro_y_dps,gyro_z_dps,fsync_tag
1715342051.234567,1205234567,1715342051234890,
  0.012,0.008,1.002,0.5,-0.2,0.1,0
1715342051.235567,1205235567,1715342051235891,
  0.015,0.010,1.005,0.6,-0.1,0.2,0
\end{verbatim}

\textbf{Column Definitions:}
\begin{description}
    \item[unified\_time\_s] Wall-clock timestamp aligned to camera frame
    \item[esp\_timestamp\_us] ESP32 hardware timestamp
    \item[host\_timestamp\_us] Host receive timestamp
    \item[accel\_x\_g ... accel\_z\_g] Accelerometer in g-force units
    \item[gyro\_x\_dps ... gyro\_z\_dps] Gyroscope in degrees/second
    \item[fsync\_tag] 1 if FSYNC edge detected, 0 otherwise
\end{description}
